    function configureQRCode() {
      const isConfigured =
        PUBLIC_URL.startsWith("https://") &&
        !PUBLIC_URL.includes("YOUR-DOMAIN");

      if (!isConfigured) {
        publicUrlElement.textContent =

        qrImage.hidden = true;
        return;
      }

      publicUrlElement.textContent = PUBLIC_URL;

      */

      const qrUrl = new URL("https://api.qrserver.com/v1/create-qr-code/");

      qrUrl.search = new URLSearchParams({

      qrImage.onload = function () {
        qrImage.hidden = false;
        qrPlaceholder.hidden = true;
      };

      qrImage.onerror = function () {
        qrImage.hidden = true;
        qrPlaceholder.hidden = false;
        qrPlaceholder.textContent = "QR code unavailable";
      };

      qrImage.src = qrUrl.toString();
    }

    /* =========================================
       3. VCF / VCARD CONTACT DOWNLOAD
    ========================================= */

    function escapeVCard(value) {
      return String(value)
        .replace(/\\/g, "\\\\")
        .replace(/\n/g, "\\n")
        .replace(/;/g, "\\;")
        .replace(/,/g, "\\,");
    }

    function downloadContact() {
      const lines = [
        "BEGIN:VCARD",
        "VERSION:3.0",
        "N:" +
          escapeVCard(contact.lastName) + ";" +
          escapeVCard(contact.firstName) + ";;;",
        "FN:" + escapeVCard(contact.fullName),
        "ORG:" + escapeVCard(contact.company),
        "TITLE:" + escapeVCard(contact.jobTitle),
        "TEL;TYPE=CELL,VOICE:" + contact.phone,
        "EMAIL;TYPE=INTERNET,WORK:" + escapeVCard(contact.email),
        "URL:" + escapeVCard(contact.website),
        "NOTE:" + escapeVCard("WhatsApp: https://wa.me/" + contact.whatsapp),
        "END:VCARD"
      ];

      const vCardContent = lines.join("\r\n") + "\r\n";

      const blob = new Blob(
        [vCardContent],
        { type: "text/vcard;charset=utf-8" }
      );

      const downloadUrl = URL.createObjectURL(blob);
      const link = document.createElement("a");

      link.href = downloadUrl;
      link.download = "Yehya-Rahhal.vcf";

      document.body.appendChild(link);
      link.click();
      link.remove();

      URL.revokeObjectURL(downloadUrl);

      showToast("Contact downloaded successfully");
    }

    document
      .getElementById("saveContact")
      .addEventListener("click", downloadContact);

       4. SHARE BUSINESS CARD
    ========================================= */

    async function shareContact() {
      const shareData = {
        title: contact.fullName + " | " + contact.company,
        text:
          contact.fullName + "\n" +
          contact.jobTitle + "\n" +
          contact.company + "\n" +
          "Phone: " + contact.phone + "\n" +
          "Email: " + contact.email,
        url: PUBLIC_URL
      };

      /*
        Use the native share menu when supported.
        Otherwise, copy the card URL to the clipboard.
      */

      if (navigator.share) {
        try {
          await navigator.share(shareData);
          return;
        } catch (error) {
          if (error.name === "AbortError") {
            return;
          }
        }
      }

      if (navigator.clipboard && window.isSecureContext) {
        try {
          await navigator.clipboard.writeText(PUBLIC_URL);
          showToast("Business card link copied");
          return;
        } catch (error) {
          // Continue to the manual copy fallback.
        }
      }

      /*
        Fallback for browsers that do not support
        Web Share or clipboard access.
      */

      const textArea = document.createElement("textarea");

      textArea.value = PUBLIC_URL;
      textArea.setAttribute("readonly", "");
      textArea.style.position = "fixed";
      textArea.style.opacity = "0";

      document.body.appendChild(textArea);
      textArea.select();

      let copied = false;

      try {
        copied = document.execCommand("copy");
      } catch (error) {
        copied = false;
      }

      textArea.remove();

      if (copied) {
        showToast("Business card link copied");
      } else {
        window.prompt("Copy your business card URL:", PUBLIC_URL);
      }
    }

    document
      .getElementById("shareContact")
      .addEventListener("click", shareContact);
